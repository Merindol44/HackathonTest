    pelvis = data.qpos[0:3]
    cam_pos = pelvis + np.array([-2.2, -1.5, 1.2])
    back = -pelvis + cam_pos; back /= np.linalg.norm(back)
    right = np.cross(np.array([0, 0, 1]), back); right /= np.linalg.norm(right)
    up = np.cross(back, right)
    data.cam_xpos[:] = cam_pos
    data.cam_xmat[:] = np.stack([right, up, back]).ravel()
